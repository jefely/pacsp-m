# src/core/credit_meter.py
import numpy as np
from typing import List, Optional
from datetime import datetime
from src.config import settings

class CreditMeter:
    @staticmethod
    def compute_raw_credit(
        kappa_curve: np.ndarray,  # shape (N, 2) -> [time, curvature]
        tau_curve: np.ndarray,    # shape (N, 2) -> [time, residence]
        transitions: Optional[List[dict]] = None
    ) -> float:
        """
        计算原始信用系数 C_raw
        对应VBA的 ComputeRawCredit 函数
        """
        if kappa_curve.shape != tau_curve.shape:
            raise ValueError("曲率与驻留曲线形状不一致")

        total = 0.0

        for i in range(len(kappa_curve) - 1):
            dt = kappa_curve[i + 1, 0] - kappa_curve[i, 0]
            if dt <= 0:
                continue

            kappa = (kappa_curve[i, 1] + kappa_curve[i + 1, 1]) / 2
            tau = (tau_curve[i, 1] + tau_curve[i + 1, 1]) / 2

            # 相变累积
            trans_sum = 0.0
            if transitions:
                for t in transitions:
                    if kappa_curve[i, 0] <= t.get("time", 0) <= kappa_curve[i + 1, 0]:
                        trans_sum += t.get("amplitude", 0) * t.get("efficiency", 0)

            total += kappa * tau * (1 + trans_sum) * dt

        return total

    @staticmethod
    def compute_effective_credit(raw_c: float, rho: float) -> float:
        """计算有效信用系数 C_eff = rho * C_raw"""
        return rho * raw_c

    @staticmethod
    def get_rho_coefficient(environment: str) -> float:
        """获取环境系数"""
        mapping = {
            "physical": 0.95,
            "human_virtual": 0.78,
            "vr_simulator": 0.62,
            "ai_virtual": 0.35,
            "data_injection": 0.0
        }
        return mapping.get(environment, 0.5)

    @staticmethod
    def aggregate_raw_credits(path_credits: List[float], correlations: np.ndarray) -> float:
        """聚合多条路径，带冗余惩罚"""
        if not path_credits:
            return 0.0

        raw_sum = sum(path_credits)
        penalty = 0.0
        eta = settings.ETA

        if correlations.size > 0:
            for i in range(len(path_credits)):
                for j in range(i + 1, len(path_credits)):
                    penalty += eta * correlations[i, j] * path_credits[i] * path_credits[j]

        return raw_sum - penalty

# src/core/cdrs_core.py
import numpy as np
from typing import List
from datetime import datetime
from src.config import settings

class CDRSCore:
    @staticmethod
    def compute_cci(gradient_norms: List[float]) -> float:
        """
        计算认知曲率指数 (CCI)
        对应VBA的 ComputeCCI
        """
        if len(gradient_norms) < 2:
            return 0.0

        arr = np.array(gradient_norms)
        avg_grad = np.mean(arr)

        # 计算梯度变化率方差
        grad_changes = np.diff(arr)
        grad_var = np.var(grad_changes)

        return avg_grad * grad_var

    @staticmethod
    def compute_rdi(loss_values: List[float], threshold: float = 0.3) -> float:
        """
        计算驻留深度指数 (RDI)
        对应VBA的 ComputeRDI
        """
        if not loss_values:
            return 0.0

        arr = np.array(loss_values)
        over_threshold = arr[arr > threshold]
        return len(over_threshold) / len(arr)

    @staticmethod
    def compute_ptbi(
        phase_transitions: List[dict],
        decay_sigma: float = 100.0
    ) -> float:
        """
        计算相变爆发指数 (PTBI)
        对应VBA的 ComputePTBI
        """
        if not phase_transitions:
            return 0.0

        total = 0.0
        now = datetime.now()

        for pt in phase_transitions:
            pt_time = pt.get("timestamp")
            jump = pt.get("perf_jump", 0)

            if pt_time:
                dt = (now - pt_time).total_seconds() / 3600  # 小时
                weight = np.exp(-dt / decay_sigma)
                total += jump * weight
            else:
                total += jump * 0.5

        return total

    @staticmethod
    def compute_cds(
        cci: float,
        rdi: float,
        ptbi: float,
        adaptability: float
    ) -> float:
        """
        计算综合深度评分 (CDS)
        对应VBA的 ComputeCDS
        """
        # 归一化 (实际生产需从历史数据动态校准)
        max_values = {
            "cci": 1000.0,
            "rdi": 0.5,
            "ptbi": 10.0,
            "adaptability": 1.2
        }

        norm_cci = min(cci / max_values["cci"], 1.0)
        norm_rdi = min(rdi / max_values["rdi"], 1.0)
        norm_ptbi = min(ptbi / max_values["ptbi"], 1.0)
        norm_adapt = min(adaptability / max_values["adaptability"], 1.0)

        w = settings.CDS_WEIGHTS
        return (w[0] * norm_cci +
                w[1] * norm_rdi +
                w[2] * norm_ptbi +
                w[3] * norm_adapt)

    @staticmethod
    def get_grade(cds: float) -> str:
        """获取评级"""
        if cds >= 0.85:
            return "S"
        elif cds >= 0.70:
            return "A"
        elif cds >= 0.50:
            return "B"
        elif cds >= 0.30:
            return "C"
        else:
            return "D"

# src/core/zkp_wrapper.py
import hashlib
import json
from typing import Optional
from datetime import datetime
from src.config import settings

class ZKPWrapper:
    """
    ZKP封装层 (生产环境中应接入真实的zk-SNARK库)
    目前使用哈希模拟验证逻辑
    """

    @staticmethod
    def generate_proof(
        agent_id: str,
        cds: float,
        salt: Optional[str] = None
    ) -> str:
        """
        模拟生成ZKP证明
        对应VBA的 GenerateProof
        """
        if salt is None:
            salt = datetime.utcnow().isoformat()

        payload = f"{agent_id}:{cds}:{salt}:ZKP_SALT"
        return f"ZKP_{hashlib.sha256(payload.encode()).hexdigest()}"

    @staticmethod
    def verify_proof(
        proof: str,
        agent_id: str,
        cds: float,
        expected_salt: Optional[str] = None
    ) -> bool:
        """
        验证ZKP证明
        对应VBA的 VerifyProof
        """
        if not proof or not proof.startswith("ZKP_"):
            return False

        if settings.ZKP_ENABLED:
            # 实际应调用zk-SNARK验证
            # 当前为模拟验证
            expected = ZKPWrapper.generate_proof(
                agent_id, cds, expected_salt or "ZKP_SALT"
            )
            return proof == expected

        return True

# src/services/rating_service.py
from typing import List, Optional
import numpy as np
from src.core.credit_meter import CreditMeter
from src.core.cdrs_core import CDRSCore
from src.core.zkp_wrapper import ZKPWrapper
from src.domain.schemas import RatingRequest, RatingResponse
from src.domain.models import Agent, CreditRecord, TrainingStep
from src.config import settings

class RatingService:
    def __init__(self, db_session=None):
        self.db = db_session
        self.credit_meter = CreditMeter()
        self.cdrs = CDRSCore()
        self.zkp = ZKPWrapper()

    async def rate_agent(self, request: RatingRequest) -> RatingResponse:
        """
        执行完整的AI深度评级
        将VBA的MainDemo流程迁移到生产环境
        """
        # 1. 提取训练步数据
        steps = request.steps
        gradient_norms = [s.gradient_norm for s in steps]
        loss_values = [s.loss_val for s in steps]

        # 2. 构建曲率和驻留曲线 (从梯度与loss推导)
        # 实际生产中可以记录更丰富的元数据
        kappa_curve = np.array([
            [i, grad_norm] for i, grad_norm in enumerate(gradient_norms)
        ])
        tau_curve = np.array([
            [i, 1.0 / (1 + loss)] for i, loss in enumerate(loss_values)
        ])

        # 3. 检测相变事件
        phase_transitions = []
        for i, step in enumerate(steps):
            # 简单检测: loss大幅下降时标记为相变
            if i > 0 and step.loss_val < steps[i-1].loss_val * 0.7:
                phase_transitions.append({
                    "timestamp": step.timestamp,
                    "perf_jump": (steps[i-1].loss_val - step.loss_val) / steps[i-1].loss_val
                })

        # 4. 计算信用系数 (UACP)
        raw_c = self.credit_meter.compute_raw_credit(
            kappa_curve, tau_curve, phase_transitions
        )
        rho = self.credit_meter.get_rho_coefficient(
            "ai_virtual" if request.agent_type == "ai" else "physical"
        )
        eff_c = self.credit_meter.compute_effective_credit(raw_c, rho)

        # 5. 计算CDRP指标
        cci = self.cdrs.compute_cci(gradient_norms)
        rdi = self.cdrs.compute_rdi(loss_values)
        ptbi = self.cdrs.compute_ptbi(phase_transitions)

        # 6. 计算泛化能力
        id_acc = request.in_distribution_accuracy or 1.0
        adaptability = (
            request.out_of_distribution_accuracy / id_acc
            if id_acc > 0 else 0
        )

        # 7. 合成CDS
        cds = self.cdrs.compute_cds(cci, rdi, ptbi, adaptability)
        grade = self.cdrs.get_grade(cds)

        # 8. 生成ZKP证明
        proof = self.zkp.generate_proof(
            request.agent_id, cds, salt=request.model_hash
        )

        # 9. 持久化 (如有数据库)
        if self.db:
            await self._persist_results(request, raw_c, eff_c, cds, grade, proof)

        return RatingResponse(
            agent_id=request.agent_id,
            cci=round(cci, 4),
            rdi=round(rdi, 4),
            ptbi=round(ptbi, 4),
            adaptability=round(adaptability, 4),
            cds=round(cds, 4),
            grade=grade,
            proof=proof
        )

# scripts/demo_rating.py
import asyncio
import httpx
import random
from datetime import datetime

async def demo_rating():
    """模拟AI训练数据并调用评级API"""

    # 生成模拟训练步数据
    steps = []
    for i in range(1000):
        steps.append({
            "step_index": i,
            "gradient_norm": 0.5 + 0.3 * (i / 100) + 0.1 * random.random(),
            "loss_val": 1.0 / (1 + i / 100) + 0.02 * random.random(),
            "timestamp": datetime.utcnow().isoformat()
        })

    request = {
        "agent_id": "test_agent_001",
        "agent_type": "ai",
        "model_hash": "llama3-8b-finetuned",
        "steps": steps,
        "in_distribution_accuracy": 0.92,
        "out_of_distribution_accuracy": 0.78
    }

    async with httpx.AsyncClient() as client:
        response = await client.post(
            "http://localhost:8000/api/v1/rating/rate",
            json=request,
            timeout=60.0
        )

        if response.status_code == 200:
            result = response.json()
            print("===== CDS深度评级结果 =====")
            print(f"Agent ID: {result['agent_id']}")
            print(f"认知曲率指数 (CCI): {result['cci']}")
            print(f"驻留深度指数 (RDI): {result['rdi']}")
            print(f"相变爆发指数 (PTBI): {result['ptbi']}")
            print(f"泛化能力: {result['adaptability']}")
            print(f"综合深度评分 (CDS): {result['cds']}")
            print(f"深度等级: {result['grade']}")
            print(f"ZKP证明: {result['proof'][:50]}...")
        else:
            print(f"Error: {response.status_code}")

if __name__ == "__main__":
    asyncio.run(demo_rating())
