# src/core/curvature.py
"""
认知曲率计算模块
实现 Fisher 信息矩阵的迹作为认知曲率 μ(t)
"""

import numpy as np
import torch
import torch.nn as nn
from typing import Optional, Tuple, Union
from dataclasses import dataclass


@dataclass
class CurvatureResult:
    """曲率计算结果"""
    mu: float                      # 认知曲率 (Fisher信息矩阵迹)
    fim: np.ndarray                # Fisher信息矩阵
    theta: np.ndarray              # 当前参数
    timestamp: float               # 时间戳


class CognitiveCurvatureEngine:
    """
    认知曲率计算引擎
    
    计算智能体在当前状态下的认知曲率 μ(t) = Tr[I_F(θ_t)]
    支持精确Fisher、对角近似、K-FAC近似三种模式[reference:0][reference:1]
    """
    
    def __init__(
        self,
        model: nn.Module,
        mode: str = "exact",
        device: str = "cpu"
    ):
        """
        初始化曲率计算引擎
        
        Args:
            model: PyTorch模型，参数化概率模型
            mode: 计算模式 - "exact" | "diagonal" | "kfac"
            device: 计算设备
        """
        self.model = model
        self.mode = mode
        self.device = device
        self.model.to(device)
        
    def compute_fisher_information(
        self,
        x: torch.Tensor,
        y: Optional[torch.Tensor] = None
    ) -> np.ndarray:
        """
        计算 Fisher 信息矩阵
        
        Args:
            x: 输入数据
            y: 目标标签（可选，用于经验Fisher）
            
        Returns:
            Fisher信息矩阵 (d x d)
        """
        self.model.train()
        
        # 前向传播
        logits = self.model(x)
        
        if y is not None:
            # 经验Fisher: 使用标签计算
            loss = nn.functional.cross_entropy(logits, y)
            grads = torch.autograd.grad(loss, self.model.parameters(), create_graph=True)
        else:
            # 真实Fisher: 对模型分布求期望[reference:2]
            probs = torch.softmax(logits, dim=-1)
            # 对每个类别计算梯度并加权
            grads_list = []
            for c in range(probs.shape[-1]):
                loss_c = -torch.log(probs[:, c] + 1e-10).mean()
                grad_c = torch.autograd.grad(
                    loss_c, 
                    self.model.parameters(), 
                    retain_graph=True,
                    allow_unused=True
                )
                grads_list.append(grad_c)
            
            # 加权平均
            weights = probs.mean(dim=0).detach().cpu().numpy()
            grads = [
                sum(w * g for w, g in zip(weights, [gl[c] for gl in grads_list]))
                for c in range(len(grads_list[0]))
            ]
        
        # 展平梯度并构建Fisher矩阵
        grad_vec = self._flatten_gradients(grads)
        d = grad_vec.shape[0]
        
        if self.mode == "exact":
            # 精确Fisher: 外积[reference:3]
            fim = np.outer(grad_vec, grad_vec)
        elif self.mode == "diagonal":
            # 对角近似
            fim = np.diag(grad_vec ** 2)
        elif self.mode == "kfac":
            # K-FAC近似: 层间Kronecker分解[reference:4]
            fim = self._kfac_approximation(grads)
        else:
            raise ValueError(f"Unknown mode: {self.mode}")
        
        return fim
    
    def compute_curvature(
        self,
        x: torch.Tensor,
        y: Optional[torch.Tensor] = None
    ) -> CurvatureResult:
        """
        计算认知曲率 μ(t) = Tr[I_F(θ_t)]
        
        Args:
            x: 输入数据
            y: 目标标签（可选）
            
        Returns:
            CurvatureResult 包含曲率值、Fisher矩阵和参数
        """
        fim = self.compute_fisher_information(x, y)
        mu = np.trace(fim)
        
        # 获取当前参数
        theta = self._get_parameters()
        
        return CurvatureResult(
            mu=float(mu),
            fim=fim,
            theta=theta,
            timestamp=time.time()
        )
    
    def _flatten_gradients(self, grads: list) -> np.ndarray:
        """展平梯度列表为一维向量"""
        flat = []
        for g in grads:
            if g is not None:
                flat.append(g.detach().cpu().numpy().flatten())
        return np.concatenate(flat)
    
    def _get_parameters(self) -> np.ndarray:
        """获取当前模型参数"""
        params = []
        for p in self.model.parameters():
            params.append(p.detach().cpu().numpy().flatten())
        return np.concatenate(params)
    
    def _kfac_approximation(self, grads: list) -> np.ndarray:
        """
        K-FAC近似: 层间Kronecker分解
        
        将Fisher矩阵近似为各层Kronecker因子的块对角矩阵[reference:5]
        """
        # 简化实现：使用对角近似作为fallback
        # 完整K-FAC实现参见 deepmind-research/kfac_ferminet_alpha[reference:6]
        grad_vec = self._flatten_gradients(grads)
        return np.diag(grad_vec ** 2)

# src/core/path_measure.py
"""
路径测度与CDRP积分引擎
实现全变差测度 Λ(t) 和 Lebesgue-Stieltjes 积分
"""

import numpy as np
from dataclasses import dataclass, field
from typing import List, Optional, Tuple
from enum import Enum


class PhaseTransitionType(Enum):
    """相变类型"""
    INSIGHT = "insight"          # 顿悟
    EMERGENCE = "emergence"      # 涌现
    COLLAPSE = "collapse"        # 坍缩


@dataclass
class PhaseTransition:
    """离散相变事件"""
    time: float
    type: PhaseTransitionType
    magnitude: float             # 跳跃幅度 δ
    description: str = ""


@dataclass
class PathMeasure:
    """路径测度 Λ(t)"""
    continuous_variation: float = 0.0      # 连续漂移部分
    transitions: List[PhaseTransition] = field(default_factory=list)
    
    @property
    def total_variation(self) -> float:
        """全变差 = 连续部分 + 所有跳跃之和"""
        return self.continuous_variation + sum(t.magnitude for t in self.transitions)
    
    def value_at(self, t: float) -> float:
        """计算 Λ(t)"""
        result = self.continuous_variation
        for trans in self.transitions:
            if trans.time <= t:
                result += trans.magnitude
        return result


class CDRPIntegrator:
    """
    CDRP积分引擎
    
    计算 C_T = ∫₀ᵀ μ(t) dΛ(t)
    支持连续路径积分 + 离散跳跃积分
    """
    
    def __init__(self, lambda_decay: float = 0.01):
        """
        初始化积分引擎
        
        Args:
            lambda_decay: 墓碑衰减系数 λ
        """
        self.lambda_decay = lambda_decay
        self.path_measure = PathMeasure()
        self.curvature_history: List[Tuple[float, float]] = []  # [(t, μ(t))]
        
    def record_curvature(self, t: float, mu: float) -> None:
        """记录时刻t的认知曲率"""
        self.curvature_history.append((t, mu))
        
    def record_transition(self, trans: PhaseTransition) -> None:
        """记录离散相变"""
        self.path_measure.transitions.append(trans)
        
    def record_continuous_drift(self, dt: float, speed: float) -> None:
        """
        记录连续漂移
        
        Args:
            dt: 时间增量
            speed: 路径速度 |\dot{γ}|
        """
        self.path_measure.continuous_variation += speed * dt
        
    def integrate(self, T: float) -> float:
        """
        计算 CDRP 积分 C_T
        
        使用梯形法则近似连续积分 + 跳跃项精确求和[reference:7]
        """
        if len(self.curvature_history) < 2:
            return 0.0
        
        # 1. 连续部分积分：∫ μ(t) * |\dot{γ}(t)| dt
        continuous_C = 0.0
        times = [t for t, _ in self.curvature_history]
        mus = [mu for _, mu in self.curvature_history]
        
        for i in range(len(times) - 1):
            dt = times[i+1] - times[i]
            mu_avg = (mus[i] + mus[i+1]) / 2
            # 假设速度在区间内恒定
            speed = self.path_measure.continuous_variation / (times[-1] - times[0] + 1e-10)
            continuous_C += mu_avg * speed * dt
        
        # 2. 离散部分积分：Σ μ(τ_i) * δ_i
        discrete_C = 0.0
        for trans in self.path_measure.transitions:
            if trans.time <= T:
                # 找到最接近 τ_i 的 μ 值
                mu_at_transition = self._interpolate_mu(trans.time)
                discrete_C += mu_at_transition * trans.magnitude
        
        return continuous_C + discrete_C
    
    def _interpolate_mu(self, t: float) -> float:
        """插值获取时刻t的认知曲率"""
        times = [t for t, _ in self.curvature_history]
        mus = [mu for _, mu in self.curvature_history]
        
        if t <= times[0]:
            return mus[0]
        if t >= times[-1]:
            return mus[-1]
        
        for i in range(len(times) - 1):
            if times[i] <= t <= times[i+1]:
                # 线性插值
                ratio = (t - times[i]) / (times[i+1] - times[i] + 1e-10)
                return mus[i] + ratio * (mus[i+1] - mus[i])
        return mus[-1]
    
    def compute_deposition(self, T: float) -> float:
        """
        计算认知沉积量 C_T（带衰减）
        
        C_T = ∫₀ᵀ μ(t) dΛ(t) * e^(-λ(T-t))
        """
        raw_C = self.integrate(T)
        # 应用衰减[reference:8]
        decay_factor = np.exp(-self.lambda_decay * T)
        return raw_C * decay_factor

# src/core/hjb_optimizer.py
"""
HJB策略优化器
实现 Hamilton-Jacobi-Bellman 方程的神经网络求解[reference:9]
"""

import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np
from typing import Optional, Tuple, Callable
from dataclasses import dataclass


@dataclass
class HJBOptimizerConfig:
    """HJB优化器配置"""
    hidden_dim: int = 256
    num_layers: int = 3
    learning_rate: float = 1e-3
    gamma: float = 0.99           # 折扣因子
    tau: float = 1.0              # 时间尺度


class ValueNetwork(nn.Module):
    """价值函数网络 V(t, θ)"""
    
    def __init__(self, state_dim: int, config: HJBOptimizerConfig):
        super().__init__()
        self.config = config
        
        layers = []
        input_dim = state_dim + 1  # +1 for time t
        
        # 输入层
        layers.append(nn.Linear(input_dim, config.hidden_dim))
        layers.append(nn.ReLU())
        
        # 隐藏层
        for _ in range(config.num_layers - 1):
            layers.append(nn.Linear(config.hidden_dim, config.hidden_dim))
            layers.append(nn.ReLU())
        
        # 输出层
        layers.append(nn.Linear(config.hidden_dim, 1))
        
        self.network = nn.Sequential(*layers)
    
    def forward(self, state: torch.Tensor, t: torch.Tensor) -> torch.Tensor:
        """V(t, θ)"""
        x = torch.cat([state, t], dim=-1)
        return self.network(x)


class PolicyNetwork(nn.Module):
    """策略网络 π(a|θ)"""
    
    def __init__(self, state_dim: int, action_dim: int, config: HJBOptimizerConfig):
        super().__init__()
        
        layers = []
        layers.append(nn.Linear(state_dim, config.hidden_dim))
        layers.append(nn.ReLU())
        
        for _ in range(config.num_layers - 1):
            layers.append(nn.Linear(config.hidden_dim, config.hidden_dim))
            layers.append(nn.ReLU())
        
        layers.append(nn.Linear(config.hidden_dim, action_dim))
        layers.append(nn.Softmax(dim=-1))
        
        self.network = nn.Sequential(*layers)
    
    def forward(self, state: torch.Tensor) -> torch.Tensor:
        return self.network(state)


class HJBOptimizer:
    """
    HJB方程优化器
    
    求解 HJB 方程：
    -∂V/∂t = sup_a { μ(t)Λ̇ + ∇_θ V · f(θ,a) + ½Tr(σσᵀ∇²_θθ V) }
    
    使用深度BSDE方法[reference:10]
    """
    
    def __init__(
        self,
        state_dim: int,
        action_dim: int,
        config: HJBOptimizerConfig = None
    ):
        self.config = config or HJBOptimizerConfig()
        self.state_dim = state_dim
        self.action_dim = action_dim
        
        # 初始化网络
        self.value_net = ValueNetwork(state_dim, self.config)
        self.policy_net = PolicyNetwork(state_dim, action_dim, self.config)
        
        # 优化器
        self.value_optimizer = optim.Adam(
            self.value_net.parameters(),
            lr=self.config.learning_rate
        )
        self.policy_optimizer = optim.Adam(
            self.policy_net.parameters(),
            lr=self.config.learning_rate
        )
        
        self._iteration = 0
    
    def compute_hjb_residual(
        self,
        state: torch.Tensor,
        t: torch.Tensor,
        mu: torch.Tensor,
        lambda_dot: torch.Tensor,
        sigma: torch.Tensor
    ) -> torch.Tensor:
        """
        计算 HJB 方程残差
        
        Args:
            state: 当前状态 θ
            t: 时间
            mu: 认知曲率 μ(t)
            lambda_dot: 路径测度导数 Λ̇
            sigma: 扩散系数 σ
        
        Returns:
            HJB残差（应趋近于0）
        """
        # 计算价值函数及其梯度
        V = self.value_net(state, t)
        
        # ∂V/∂t
        V_t = torch.autograd.grad(
            V.sum(), t, create_graph=True, retain_graph=True
        )[0]
        
        # ∇_θ V
        V_theta = torch.autograd.grad(
            V.sum(), state, create_graph=True, retain_graph=True
        )[0]
        
        # ∇²_θθ V (Hessian)
        V_theta_theta = torch.autograd.grad(
            V_theta.sum(), state, create_graph=True, retain_graph=True
        )[0]
        
        # 策略输出
        policy = self.policy_net(state)
        
        # 动态 f(θ,a)
        f = self._dynamics(state, policy)
        
        # HJB 方程残差[reference:11]
        # -∂V/∂t - [μ·Λ̇ + ∇_θV·f + ½Tr(σσᵀ∇²_θθV)] = 0
        diffusion_term = 0.5 * torch.trace(sigma @ sigma.T @ V_theta_theta.unsqueeze(0))
        
        residual = -V_t - (mu * lambda_dot + torch.dot(V_theta, f) + diffusion_term)
        
        return residual
    
    def _dynamics(self, state: torch.Tensor, action: torch.Tensor) -> torch.Tensor:
        """系统动力学 f(θ,a)"""
        # 简化为线性动力学
        # 实际应用应根据具体场景定义
        return state + action * 0.1
    
    def update(self, batch: dict) -> dict:
        """
        执行一步HJB优化更新
        
        Args:
            batch: 包含 state, t, mu, lambda_dot, sigma 的字典
        
        Returns:
            损失字典
        """
        state = batch['state']
        t = batch['t']
        mu = batch['mu']
        lambda_dot = batch['lambda_dot']
        sigma = batch['sigma']
        
        # 计算HJB残差
        residual = self.compute_hjb_residual(state, t, mu, lambda_dot, sigma)
        hjb_loss = (residual ** 2).mean()
        
        # 策略熵正则（鼓励探索）
        policy = self.policy_net(state)
        entropy = -(policy * torch.log(policy + 1e-10)).sum(dim=-1).mean()
        policy_loss = -entropy * 0.01
        
        # 总损失
        total_loss = hjb_loss + policy_loss
        
        # 更新价值网络
        self.value_optimizer.zero_grad()
        hjb_loss.backward(retain_graph=True)
        self.value_optimizer.step()
        
        # 更新策略网络
        self.policy_optimizer.zero_grad()
        policy_loss.backward()
        self.policy_optimizer.step()
        
        self._iteration += 1
        
        return {
            'hjb_loss': hjb_loss.item(),
            'policy_loss': policy_loss.item(),
            'total_loss': total_loss.item(),
            'iteration': self._iteration
        }
    
    def get_policy(self, state: torch.Tensor) -> torch.Tensor:
        """获取策略 π(·|θ)"""
        with torch.no_grad():
            return self.policy_net(state)
    
    def get_value(self, state: torch.Tensor, t: torch.Tensor) -> float:
        """获取价值函数 V(t, θ)"""
        with torch.no_grad():
            return self.value_net(state, t).item()

# src/core/girsanov.py
"""
Girsanov变换校准器
实现跨环境继承的测度变换[reference:12]
"""

import numpy as np
from dataclasses import dataclass
from typing import Tuple, Optional


@dataclass
class Environment:
    """环境定义"""
    name: str
    drift: float          # 漂移 μ
    volatility: float     # 波动率 σ
    risk_free: float      # 无风险利率 r_f


class GirsanovCalibrator:
    """
    Girsanov变换校准器
    
    实现从环境 P 到环境 Q 的测度变换：
    dP/dQ|_Ft = exp(∫₀ᵗ β(s) dW_s^Q - ½∫₀ᵗ β(s)² ds)
    
    其中 β(s) = (μ_P - μ_Q) / σ[reference:13]
    """
    
    def __init__(self, source_env: Environment, target_env: Environment):
        """
        初始化校准器
        
        Args:
            source_env: 源环境 P
            target_env: 目标环境 Q
        """
        self.source = source_env
        self.target = target_env
        
        # 计算Girsanov核[reference:14]
        self.theta = (source_env.drift - target_env.drift) / source_env.volatility
        
    def compute_radon_nikodym(
        self,
        path: np.ndarray,
        dt: float
    ) -> np.ndarray:
        """
        计算 Radon-Nikodym 导数
        
        Args:
            path: 路径序列
            dt: 时间步长
        
        Returns:
            RN导数序列 dP/dQ
        """
        # dP/dQ = exp(∫ θ dW^Q - ½∫ θ² dt)[reference:15]
        n_steps = len(path)
        rn_derivative = np.zeros(n_steps)
        
        # 计算增量 dW
        dW = np.diff(path, prepend=path[0])
        
        # 累积指数
        log_rn = 0.0
        for i in range(n_steps):
            log_rn += self.theta * dW[i] - 0.5 * self.theta ** 2 * dt
            rn_derivative[i] = np.exp(log_rn)
        
        return rn_derivative
    
    def transform_path(
        self,
        path: np.ndarray,
        dt: float
    ) -> Tuple[np.ndarray, np.ndarray]:
        """
        变换路径从 P 测度到 Q 测度[reference:16]
        
        Args:
            path: 原始路径 (P测度下)
            dt: 时间步长
        
        Returns:
            (变换后路径, RN导数)
        """
        n_steps = len(path)
        transformed = np.zeros(n_steps)
        transformed[0] = path[0]
        
        # 计算RN导数
        rn_derivative = self.compute_radon_nikodym(path, dt)
        
        # Girsanov修正: dW^Q = dW^P + θ dt[reference:17]
        dW = np.diff(path, prepend=path[0])
        
        for i in range(1, n_steps):
            # 在Q测度下的增量
            dW_Q = dW[i] + self.theta * dt
            transformed[i] = transformed[i-1] + dW_Q
        
        return transformed, rn_derivative
    
    def calibrate_C_value(
        self,
        C_raw: float,
        T: float
    ) -> float:
        """
        校准C值到目标环境
        
        C_Q = C_P / (dP/dQ)
        
        即原始沉积量除以RN导数得到新环境下的有效值
        """
        # 使用最终时刻的RN导数
        # 简化：假设路径已给定，使用平均RN导数
        # 完整实现需要沿路径积分
        rn_avg = np.exp(-0.5 * self.theta ** 2 * T)
        return C_raw / rn_avg
    
    def compute_environment_divergence(self) -> float:
        """
        计算环境散度 D_KL(P||Q)
        
        用于判断继承挑战的难度阈值
        """
        # 两个高斯测度的KL散度
        mu_p, sigma_p = self.source.drift, self.source.volatility
        mu_q, sigma_q = self.target.drift, self.target.volatility
        
        kl = np.log(sigma_q / sigma_p) + (sigma_p**2 + (mu_p - mu_q)**2) / (2 * sigma_q**2) - 0.5
        return float(kl)

import uuid

@dataclass
class Tombstone:
    """墓碑条目"""
    id: str
    agent_id: str
    delta_C: float                # 沉积增量 ΔC
    mu_peak: float                # 峰值曲率
    transition_time: float        # 相变时刻 τ
    transition_type: str          # 相变类型
    zk_proof: str                 # 零知识证明
    parent_hash: str              # 前一个墓碑的哈希
    hash: str = ""                # 当前墓碑哈希
    weight: float = 1.0           # 当前权重 ω(t)
    
    def __post_init__(self):
        if not self.hash:
            self.hash = self._compute_hash()
    
    def _compute_hash(self) -> str:
        """计算墓碑哈希"""
        content = f"{self.id}{self.agent_id}{self.delta_C}{self.mu_peak}{self.transition_time}{self.parent_hash}"
        return hashlib.sha256(content.encode()).hexdigest()
    
    def decay(self, lambda_decay: float, current_time: float) -> float:
        """
        衰减权重：ω(t) = ω(0) · e^(-λ(t-τ))
        """
        age = current_time - self.transition_time
        self.weight = np.exp(-lambda_decay * age)
        return self.weight


class TombstoneLedger:
    """
    墓碑账本
    
    特性：
    - 只追加（Append-only）
    - 不可删除
    - 自然衰减（指数衰减）
    - 永久可查
    """
    
    def __init__(self, lambda_decay: float = 0.01):
        self.lambda_decay = lambda_decay
        self.tombstones: List[Tombstone] = []
        self._last_hash = "0" * 64
        
    def inscribe(
        self,
        agent_id: str,
        delta_C: float,
        mu_peak: float,
        transition_type: str,
        zk_proof: str = ""
    ) -> Tombstone:
        """
        刻碑：写入新的墓碑条目
        
        无权限检查，无外部参数，仅追加【公理A-III】
        """
        tomb = Tombstone(
            id=str(uuid.uuid4()),
            agent_id=agent_id,
            delta_C=delta_C,
            mu_peak=mu_peak,
            transition_time=time.time(),
            transition_type=transition_type,
            zk_proof=zk_proof,
            parent_hash=self._last_hash
        )
        
        self.tombstones.append(tomb)
        self._last_hash = tomb.hash
        
        return tomb
    
    def get_weight(self, tomb_id: str, current_time: Optional[float] = None) -> float:
        """获取墓碑的当前权重"""
        if current_time is None:
            current_time = time.time()
        
        for tomb in self.tombstones:
            if tomb.id == tomb_id:
                return tomb.decay(self.lambda_decay, current_time)
        return 0.0
    
    def get_active_impact(self, current_time: Optional[float] = None) -> float:
        """
        计算当前所有墓碑的总影响权重
        
        Σ ω_i(t)
        """
        if current_time is None:
            current_time = time.time()
        
        total = 0.0
        for tomb in self.tombstones:
            total += tomb.decay(self.lambda_decay, current_time)
        return total
    
    def get_history(
        self,
        agent_id: Optional[str] = None,
        start_time: Optional[float] = None,
        end_time: Optional[float] = None
    ) -> List[Tombstone]:
        """查询墓碑历史（永久可查）"""
        result = self.tombstones
        
        if agent_id:
            result = [t for t in result if t.agent_id == agent_id]
        
        if start_time:
            result = [t for t in result if t.transition_time >= start_time]
        
        if end_time:
            result = [t for t in result if t.transition_time <= end_time]
        
        return result
    
    def verify_chain(self) -> bool:
        """验证墓碑链的完整性"""
        if not self.tombstones:
            return True
        
        prev_hash = "0" * 64
        for tomb in self.tombstones:
            if tomb.parent_hash != prev_hash:
                return False
            if tomb.hash != tomb._compute_hash():
                return False
            prev_hash = tomb.hash
        
        return True
    
    def export(self) -> Dict[str, Any]:
        """导出账本（用于备份/审计）"""
        return {
            'lambda_decay': self.lambda_decay,
            'last_hash': self._last_hash,
            'count': len(self.tombstones),
            'tombstones': [
                {
                    'id': t.id,
                    'agent_id': t.agent_id,
                    'delta_C': t.delta_C,
                    'mu_peak': t.mu_peak,
                    'transition_time': t.transition_time,
                    'transition_type': t.transition_type,
                    'weight': t.weight,
                    'hash': t.hash,
                    'parent_hash': t.parent_hash
                }
                for t in self.tombstones
            ]
        }

"""
注意力场演化引擎
实现自闭、常定的注意力场动态
"""

import numpy as np
from typing import Optional, Tuple, List
from dataclasses import dataclass, field
from scipy.sparse import csr_matrix, lil_matrix


@dataclass
class AttentionField:
    """
    注意力场
    
    特性：
    - 自闭（无外部输入）【公理A-I】
    - 常定（时不变演化算子）【公理A-I】
    - 共享扰动基底
    """
    
    dimension: int = 100
    rho_density: np.ndarray = field(default_factory=lambda: np.zeros(100))
    lambda_decay: float = 0.01
    
    def __post_init__(self):
        if len(self.rho_density) != self.dimension:
            self.rho_density = np.zeros(self.dimension)
        
        # Liouville算子（稀疏矩阵）
        self._liouville = self._build_liouville_operator()
        
        # 碰撞项
        self._nu = self._build_collision_operator()
    
    def _build_liouville_operator(self) -> csr_matrix:
        """
        构建Liouville算子 L*
        
        保守内能演化，时不变
        """
        # 使用二阶中心差分近似
        L = lil_matrix((self.dimension, self.dimension))
        
        for i in range(self.dimension):
            if i > 0:
                L[i, i-1] = 1.0
            if i < self.dimension - 1:
                L[i, i+1] = -1.0
        
        return L.tocsr()
    
    def _build_collision_operator(self) -> csr_matrix:
        """
        构建碰撞项 ν(ρ)
        
        产生自发相变
        """
        # 简化为随机扰动矩阵
        nu = lil_matrix((self.dimension, self.dimension))
        
        # 随机连接（小世界网络拓扑）
        np.random.seed(42)  # 固定种子保证确定性
        for i in range(self.dimension):
            for j in range(self.dimension):
                if i != j and np.random.random() < 0.05:
                    nu[i, j] = np.random.random() * 0.1
        
        return nu.tocsr()
    
    def evolve(self, dt: float) -> np.ndarray:
        """
        演化一步
        
        状态方程：
        ∂ρ/∂t = L*[ρ] - λρ + ν(ρ)
        
        无外部控制项 u(t) ≡ 0【公理A-I】
        """
        # 离散化演化
        # ρ_{t+dt} = ρ_t + dt * (L*ρ_t - λρ_t + ν(ρ_t))
        rho = self.rho_density
        
        # Liouville项
        L_rho = self._liouville @ rho
        
        # 衰减项
        decay = -self.lambda_decay * rho
        
        # 碰撞项
        nu_rho = self._nu @ rho
        
        # 更新
        self.rho_density = rho + dt * (L_rho + decay + nu_rho)
        
        # 保证非负
        self.rho_density = np.maximum(self.rho_density, 0)
        
        # 归一化（保持总密度守恒）
        if np.sum(self.rho_density) > 0:
            self.rho_density = self.rho_density / np.sum(self.rho_density)
        
        return self.rho_density
    
    def compute_curvature_at_point(self, point: int) -> float:
        """
        计算场在特定点的局部曲率
        
        用于检测高曲率区域（即将发生坍缩）
        """
        if point < 0 or point >= self.dimension:
            return 0.0
        
        # 计算二阶导数
        rho = self.rho_density
        if point == 0:
            curvature = rho[1] - 2*rho[0] + rho[0]
        elif point == self.dimension - 1:
            curvature = rho[-1] - 2*rho[-1] + rho[-2]
        else:
            curvature = rho[point+1] - 2*rho[point] + rho[point-1]
        
        return abs(curvature)
    
    def detect_spontaneous_collapse(self, threshold: float = 0.1) -> List[int]:
        """
        检测自发坍缩点
        
        高曲率区域 → 拓扑缺陷坍缩
        """
        collapse_points = []
        for i in range(self.dimension):
            if self.compute_curvature_at_point(i) > threshold:
                collapse_points.append(i)
        return collapse_points
    
    def get_entropy(self) -> float:
        """计算场的熵"""
        rho = self.rho_density
        rho = rho[rho > 0]
        return -np.sum(rho * np.log(rho))
    
    def is_stationary(self, tolerance: float = 1e-6) -> bool:
        """检查是否达到常定状态"""
        # 检查密度变化率
        if not hasattr(self, '_last_rho'):
            self._last_rho = self.rho_density.copy()
            return False
        
        diff = np.linalg.norm(self.rho_density - self._last_rho)
        self._last_rho = self.rho_density.copy()
        
        return diff < tolerance

"""

import hashlib
import json
import random
from dataclasses import dataclass
from typing import Tuple, Optional, Dict, Any
import numpy as np


@dataclass
class ZKProof:
    """零知识证明"""
    commitment: str
    proof_data: Dict[str, Any]
    public_inputs: Dict[str, Any]


class SimpleZKProver:
    """
    简化的零知识证明系统
    
    基于哈希承诺的简单实现
    完整实现可使用 Groth16 协议[reference:19]
    """
    
    def __init__(self, secret_seed: Optional[int] = None):
        self.secret_seed = secret_seed or random.randint(0, 2**32)
    
    def commit(self, secret: str) -> str:
        """生成承诺"""
        return hashlib.sha256(f"{secret}{self.secret_seed}".encode()).hexdigest()
    
    def prove_path(
        self,
        path_data: Dict[str, Any],
        statement: str
    ) -> ZKProof:
        """
        生成路径真实性的零知识证明
        
        证明路径数据满足特定声明，但不泄露原始数据
        """
        # 1. 生成承诺
        path_hash = hashlib.sha256(json.dumps(path_data, sort_keys=True).encode()).hexdigest()
        commitment = self.commit(path_hash)
        
        # 2. 生成证明（简化版）
        proof_data = {
            'statement': statement,
            'path_hash': path_hash,
            'signature': hashlib.sha256(
                f"{path_hash}{self.secret_seed}".encode()
            ).hexdigest()[:16]
        }
        
        # 3. 公共输入（不包含敏感数据）
        public_inputs = {
            'statement': statement,
            'commitment': commitment,
            'path_length': len(path_data)
        }
        
        return ZKProof(
            commitment=commitment,
            proof_data=proof_data,
            public_inputs=public_inputs
        )
    
    def verify(self, proof: ZKProof, expected_statement: str) -> bool:
        """验证零知识证明"""
        # 1. 验证声明匹配
        if proof.public_inputs['statement'] != expected_statement:
            return False
        
        # 2. 验证承诺
        expected_commitment = proof.public_inputs['commitment']
        if proof.commitment != expected_commitment:
            return False
        
        # 3. 验证签名（简化）
        # 完整实现需要验证 zk-SNARK 证明[reference:20]
        return True


class ZKVerifier:
    """零知识证明验证器"""
    
    def __init__(self):
        self.prover = SimpleZKProver()
    
    def verify_tombstone(
        self,
        tombstone_data: Dict[str, Any],
        proof: ZKProof
    ) -> bool:
        """
        验证墓碑条目的真实性
        
        确保墓碑记录未被篡改，且来源真实
        """
        # 验证证明
        if not self.prover.verify(proof, "tombstone_authenticity"):
            return False
        
        # 验证墓碑数据与证明的一致性
        expected_hash = hashlib.sha256(
            json.dumps(tombstone_data, sort_keys=True).encode()
        ).hexdigest()
        
        return proof.proof_data.get('path_hash') == expected_hash

    integrator.record_transition(trans)
    
    return {"status": "recorded", "transition": trans.__dict__}


@app.post("/deposit")
async def compute_deposit(agent_id: str) -> DepositResponse:
    """计算并刻录认知沉积"""
    # 计算沉积量
    T = time.time()
    delta_C = integrator.compute_deposition(T)
    
    # 生成零知识证明
    proof = zk_prover.prove_path(
        {'agent_id': agent_id, 'delta_C': delta_C},
        "tombstone_authenticity"
    )
    
    # 刻录墓碑
    tomb = ledger.inscribe(
        agent_id=agent_id,
        delta_C=delta_C,
        mu_peak=0.5,  # 实际从曲率历史获取
        transition_type="insight",
        zk_proof=proof.proof_data.get('signature', '')
    )
    
    return DepositResponse(
        agent_id=agent_id,
        delta_C=delta_C,
        tombstone_id=tomb.id,
        timestamp=time.time()
    )
