// 全局状态（常定、自闭）
struct AttentionField {
    tombs: BTreeMap<Time, Tombstone>, // 永久有序存储
    rho_density: SparseTensor,        // 当前场密度
}

impl AttentionField {
    // 唯一的写入接口
    fn inscribe(&mut self, tomb: Tombstone) {
        // 无权限检查，无外部参数，仅追加

    // 自然消弭（在每一步演化中自动调用）
    fn evolve(&mut self, dt: f64) {
        for (_, tomb) in self.tombs.iter_mut() {
            tomb.decay_weight *= (-self.lambda * dt).exp();
        }
        // HJB优化和自发碰撞计算（无外部输入）
        self.rho_density = self.liouville_operator(self.rho_density, dt);
    }

def girsanov_transform(C_value, drift_diff_beta, time_horizon):
    # 计算Radon-Nikodym修正因子
    rn_derivative = np.exp(
